```mermaid
    USERS {
        bigint id PK "1.1 Primary Key"
        string name "Staff Member Name"
        string username UK "Staff Badge / Username"
        string email UK "Work Email"
        enum role "staff | operator | prepress_designer | qc_lead"
        string avatar "Profile Avatar URL"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    STAFF_PROFILES {
        bigint id PK "1.2 Primary Key"
        bigint user_id FK "References 1.1 USERS"
        string employee_id UK "STF-2026-001"
        string department "Prepress | Printing | Heat Press | QC | Finishing"
        string position "Machine Operator | Graphic Artist | QC Inspector"
        string workstation "Assigned Station (UV #2, DTF Line #1)"
        string shift "Morning (6AM-2PM) | Afternoon (2PM-10PM) | Night"
        boolean is_active "Active On-Duty Flag"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    TASKS {
        bigint id PK "1.3 Primary Key"
        bigint user_id FK "Assigned Staff 1.1 USERS"
        string title "Work / Operation Title"
        text description "Detailed Work Directives & SOP"
        enum priority "low | medium | high | urgent"
        enum status "pending | in_progress | completed | cancelled"
        timestamp due_date "Target Completion Timestamp"
        timestamp completed_at "Finished Timestamp"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    ORDER_ITEMS {
        bigint id PK "2.1 Primary Key"
        bigint order_id FK "References Parent Order"
        string product_name "Snapshot Product Name"
        string sku "SKU Identifier"
        string selected_size "Size Specification (S, M, L, XL)"
        string selected_color "Color Hex / Name"
        int quantity "Batch Quantity Ordered"
        decimal unit_price "Unit Price PHP"
        decimal customization_fee "Print Surcharge PHP"
        decimal line_total "Line Item Total PHP"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    ORDER_CUSTOMIZATIONS {
        bigint id PK "2.2 Primary Key"
        bigint order_item_id FK "References 2.1 ORDER_ITEMS"
        string print_location "Front | Back | Left Sleeve | Full Wrap"
        string print_technique "DTF | UV Flatbed | Sublimation | Decal"
        string custom_text "Engraved / Printed Text"
        string font_family "Typography Font Name"
        string text_color_hex "Ink Hex Code (#FFFFFF)"
        string uploaded_artwork_url "High-Res Vector / PNG (300 DPI)"
        string preview_render_url "3D Customizer Snapshot URL"
        json transform_data "3D Matrix {scale, rot, pos_x, pos_y, pos_z}"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    PROOF_APPROVALS {
        bigint id PK "3.1 Primary Key"
        bigint order_item_id FK "References 2.1 ORDER_ITEMS"
        bigint staff_profile_id FK "References 1.2 STAFF_PROFILES"
        string proof_version "v1 | v2 | v3"
        string proof_pdf_url "Print-Ready Proof PDF / Preview URL"
        enum status "pending_review | approved | changes_requested"
        text staff_remarks "Prepress Operator Layout Notes"
        text customer_feedback "Client Change Annotations"
        timestamp sent_at "Sent to Client Timestamp"
        timestamp reviewed_at "Client Decision Timestamp"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    PROOF_REVISION_MESSAGES {
        bigint id PK "3.2 Primary Key"
        bigint proof_approval_id FK "References 3.1 PROOF_APPROVALS"
        enum sender_type "staff | customer"
        bigint sender_id "User / Customer ID"
        text message "Revision Feedback / Clarification"
        string attachment_url "Annotation Markup Image"
        timestamp created_at "Message Timestamp"
    }

    PRODUCTION_JOBS {
        bigint id PK "4.1 Primary Key"
        bigint order_item_id FK "References 2.1 ORDER_ITEMS"
        bigint staff_profile_id FK "References 1.2 STAFF_PROFILES"
        string job_ticket_number UK "JOB-2026-0001"
        enum stage "queued | rip_processing | printing | heat_press | qc_inspection | packed"
        enum priority "normal | rush | vip_express"
        string assigned_machine "Assigned Machine Code"
        text operator_notes "Execution & RIP Settings Log"
        timestamp started_at "Print Start Time"
        timestamp finished_at "Curing Done Time"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    MACHINE_STATIONS {
        bigint id PK "4.2 Primary Key"
        string machine_code UK "MCH-UV-02"
        string machine_name "Roland TrueVIS / Mimaki UV / Epson DTF"
        string print_method "DTF | UV Flatbed | Sublimation"
        enum status "idle | active | maintenance | offline"
        timestamp last_maintenance_at "Last Service Date"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    PRINT_ITEMS {
        bigint id PK "5.1 Primary Key"
        string item_code UK "PRN-001"
        string name "Substrate / Print Service"
        string paper_type "Glossy | Matte | Synthetic Vinyl | Canvas"
        string paper_size "Letter | Legal | A3 | Custom Roll SqFt"
        enum print_sides "single_sided | double_sided"
        enum color_mode "full_color | grayscale | spot_white"
        string turnaround_time "Same Day | 1-2 Days"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    INVENTORY_MATERIALS {
        bigint id PK "5.2 Primary Key"
        bigint print_item_id FK "References 5.1 PRINT_ITEMS"
        string material_code UK "MAT-VINYL-01"
        string name "Raw Material Stock Name"
        string brand "Supplier Brand"
        string model "Material Series"
        int available_quantity "Current Stock Count"
        string unit "rolls | sheets | liters | pcs"
        int reorder_point "Minimum Safety Reorder Point"
        decimal unit_cost "Procurement Unit Cost PHP"
        string status "in_stock | low_stock | depleted"
        timestamp created_at "Created Timestamp"
        timestamp updated_at "Updated Timestamp"
    }

    QC_INSPECTION_LOGS {
        bigint id PK "5.3 Primary Key"
        bigint production_job_id FK "References 4.1 PRODUCTION_JOBS"

        boolean alignment_passed "Registration & Bleed Check"

        enum inspection_result "passed | rework_required | rejected_scrap"
        text remarks "QC Notes & Sign-off Details"
        timestamp inspected_at "Inspection Timestamp"
    }
```

```
                     │                                   │
                     ▼                                   ▼

                     │                                   │
                     └─────────────────┬─────────────────┘
                                       ▼
 [Level 3]                    3.1 PROOF_APPROVALS
                                       │ ◄─── 3.2 PROOF_REVISION_MESSAGES
                                       ▼
 [Level 4]                    4.1 PRODUCTION_JOBS
                                       │ ◄─── 4.2 MACHINE_STATIONS
                                       ▼
 [Level 5]                    5.1 PRINT_ITEMS (Substrates)
                                       │ ────► 5.2 INVENTORY_MATERIALS (Stock Deduct)
                                       ▼
 [Level 6]                    5.3 QC_INSPECTION_LOGS (Quality Assurance Sign-off)
```
